let MOTHS = [], mothAcc = 0;
function lampSpots() { const out = []; if (LT.lamp * LT.dark > .05) { const dx = Math.sin(R.psi), dy = -Math.cos(R.psi); for (const u of [.25, .45, .7]) out.push([R.x + dx * 60 * u * 2.4, R.y - 12 + dy * 60 * u * 2.4, 14 + u * 10]) }
  for (const S of LIGHTS) if (Math.abs(S.x - R.x) < W && Math.abs(S.y - R.y) < H && lightLevel(S, LT.dark) > .3) out.push([S.x, S.y - (S.h || 20), 10]); return out }
function stepMoths(dt) { const act = clamp((LT.dark - .3) / .25, 0, 1) * (1 - clamp((WX.rain - .15) / .3, 0, 1)) * (1 - clamp((WIND.s - .45) / .3, 0, 1)) * (1 - .6 * FROST);
  mothAcc = Math.min(3, mothAcc + dt * act * .8); const spots = act > 0 ? lampSpots() : [];
  while (mothAcc >= 1 && MOTHS.length < 12 && spots.length) { mothAcc--; const sp = spots[Math.floor(Math.random() * spots.length)], a = Math.random() * TAU;
    MOTHS.push({ x: sp[0] + Math.cos(a) * 80, y: sp[1] + Math.sin(a) * 60 - 20, sp, t: 0, dur: 2 + Math.random() * 4, r: 3 + Math.random() * 7, ph: Math.random() * 6, st: 0, c: Math.random() < .5 ? [232, 222, 196] : [214, 200, 170] }) }
  for (const m of MOTHS) { m.t += dt; let tx, ty;
    if (m.st === 0) { tx = m.sp[0]; ty = m.sp[1]; if (Math.hypot(tx - m.x, ty - m.y) < 8) { m.st = 1; m.t = 0 } else if (m.t > 6) { m.st = 2; m.t = 0 } }
    else if (m.st === 1) { const w = m.t * 2.3 + m.ph; tx = m.sp[0] + Math.cos(w) * m.r + Math.sin(m.t * 5.1) * 2; ty = m.sp[1] + Math.sin(w * 1.3) * m.r * .7; if (m.t > m.dur || act < .05) { m.st = 2; m.t = 0; m.out = Math.random() * TAU } }
    else { tx = m.x + Math.cos(m.out) * 40; ty = m.y + Math.sin(m.out) * 40 }
    const dx = tx - m.x, dy = ty - m.y, l = Math.hypot(dx, dy) || 1, v = Math.min(l, (m.st === 1 ? 22 : 30) * dt); m.x += dx / l * v + (Math.random() - .5) * 1.2; m.y += dy / l * v + (Math.random() - .5) * 1.2 }
  MOTHS = MOTHS.filter(m => !(m.st === 2 && m.t > 2)) }
// ---- foxfire: honey-fungus mycelium on old wet wood glows a faint green in the dark ----
const isFox = o => (o.t === 'log' || o.t === 'stump' || o.t === 'snag') && (o.seed % 10) < 4;
function foxfire(L, dark) { if (dark < .45) return; const a = .16 * clamp((dark - .45) / .3, 0, 1) * (1 + .2 * WET);
  for (const o of VIS) { if (!isFox(o)) continue; const fl = .8 + .2 * Math.sin(t * .7 + o.seed % 31); for (let k = 0; k < 3; k++) { const px = o.x + (h2(o.seed, k, 71) - .5) * Math.min(20, o.mw * .6), py = o.y - 1 - h2(o.seed, k, 72) * 4, g = L.createRadialGradient(px, py, 0, px, py, 5);
      g.addColorStop(0, `rgba(120,255,150,${a * fl})`); g.addColorStop(1, 'rgba(120,255,150,0)'); L.fillStyle = g; L.fillRect(px - 5, py - 5, 10, 10) } } }
function drawFoxDots() { if (LT.dark < .45) return; const a = clamp((LT.dark - .45) / .3, 0, 1); for (const o of VIS) { if (!isFox(o)) continue; for (let k = 0; k < 7; k++) { if (h2(o.seed, k, 73 + Math.floor(t * .5)) < .3) continue; ctx.fillStyle = `rgba(150,255,170,${.55 * a})`;
      ctx.fillRect(Math.round(o.x + (h2(o.seed, k, 71) - .5) * Math.min(20, o.mw * .6)), Math.round(o.y - 1 - h2(o.seed, k, 74) * 5), 1, 1) } } }
